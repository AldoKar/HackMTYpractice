// Set up

void setup() {
    Serial.begin(9600);
    delay(500);
    
    // Inicializar I2C
    Wire.begin(21, 22); // SDA=21, SCL=22
    
    // Conectar WiFi 
    Serial.print("Conectando a WiFi ");
    Serial.println(ssid);
    WiFi.mode(WIFI_STA);
    WiFi.begin(ssid, password);
    
    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
        Serial.print(".");
    }
    Serial.println("\nWiFi conectado");
    Serial.print("IP local: ");
    Serial.println(WiFi.localIP());
    Serial.println();
    
    // Configurar NTP y obtener geolocalización gracias a la API de Google Cloud

    setClock();
    obtenerGeolocalizacion();
    
    // Escanear bus I2C 
  
    byte dispositivos = 0;
    for (byte i = 1; i < 127; i++) {
        Wire.beginTransmission(i);
        if (Wire.endTransmission() == 0) {
            Serial.print("Dispositivo en 0x");
            if (i < 16) Serial.print('0');
            Serial.println(i, HEX);
            dispositivos++;
        }
    }
    
    if (dispositivos == 0) {
        Serial.println("ERROR: No se encontraron dispositivos I2C");
        while (1) delay(10);
    }
    Serial.print("Total: ");
    Serial.println(dispositivos);
    Serial.println();
    
    // Inicializar RTC DS3231 
    Serial.println("Inicializando RTC DS3231");
    if (!rtc.begin()) {
        Serial.println("ERROR: No se encontro el RTC");
        while (1) delay(10);
    }
    
    if (rtc.lostPower()) {
        Serial.println("RTC perdio energia, ajustando...");
        rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
    }
    Serial.println("RTC inicializado OK");
    
    // Inicializar MPU6050 
    Serial.println("\n Inicializando MPU6050");
    if (!mpu.begin(0x69)) { 
        if (!mpu.begin(0x68)) {
            Serial.println("ERROR: No se encontro MPU6050");
            while (1) delay(10);
        } else {
            Serial.println("MPU6050 encontrado en 0x68");
        }
    } else {
        Serial.println("MPU6050 encontrado en 0x69 OK");
    }
    
    // Configurar rangos del MPU6050
    mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
    mpu.setGyroRange(MPU6050_RANGE_500_DEG);
    mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);
    Serial.println("MPU6050 configurado");
    
    Serial.println("\n=================================");
    Serial.println("Sistema listo");
    Serial.println("=================================\n");
    
    delay(100);
}

// Loop

void loop() {
    // La lógica principal se ejecuta dentro de serialOn/serialOff,
    // garantizando que las lecturas y los envíos se hagan en cada ciclo.
    if (mostrarSerial) {
        serialOn();
    } else {
        serialOff();
    }
}



// FUNCIONES DE ENVÍO Y LÓGICA


// Configurar reloj NTP (necesario para geolocalización)
void setClock() {
    configTime(0, 0, "pool.ntp.org", "time.nist.gov");
    time_t now = time(nullptr);
    while (now < 8 * 3600 * 2) {
        delay(200);
        now = time(nullptr);
    }
}

// Obtener geolocalización WiFi
void obtenerGeolocalizacion() {
    location_t loc = location.getGeoFromWiFi();
    latitud = loc.lat;
    longitud = loc.lon;
}

// Envía datos al servidor local (Antigua función 'enviarDatos')
void enviarDatosLocal(DateTime now, float at, float T, float safecoin) {
    if (WiFi.status() == WL_CONNECTED) {
        HTTPClient http;
        http.begin(serverUrl); // Usa la URL del servidor local
        http.addHeader("Content-Type", "application/json");
        
        // Formatear fecha y hora
        String fecha = String(now.year()) + "-" + 
                       (now.month() < 10 ? "0" : "") + String(now.month()) + "-" +
                       (now.day() < 10 ? "0" : "") + String(now.day());
        
        String hora = (now.hour() < 10 ? "0" : "") + String(now.hour()) + ":" +
                      (now.minute() < 10 ? "0" : "") + String(now.minute()) + ":" +
                      (now.second() < 10 ? "0" : "") + String(now.second());
        
        // Crear JSON con ArduinoJson
        const size_t CAPACITY = JSON_OBJECT_SIZE(8); 
        StaticJsonDocument<CAPACITY> doc;

        doc["at"] = serialized(String(at, 2));
        doc["T"] = serialized(String(T, 1));
        doc["fecha"] = fecha;
        doc["hora"] = hora;
        doc["timestamp"] = now.unixtime();
        doc["latitud"] = serialized(String(latitud, 7));
        doc["longitud"] = serialized(String(longitud, 7));
        doc["safecoin"] = serialized(String(safecoin, 2));
        
        String jsonData;
        serializeJson(doc, jsonData);
        
        Serial.println("\nEnviando a Servidor Local: " + jsonData);
        
        int code = http.POST(jsonData);
        if (code > 0) {
            Serial.println("Datos enviados a LOCAL OK - Codigo: " + String(code));
        } else {
            Serial.println("Error al enviar a LOCAL - Codigo: " + String(code));
            Serial.println("Respuesta del servidor: " + http.getString());
        }
        
        http.end();
    } else {
        Serial.println("WiFi desconectado - No se enviaron datos al LOCAL");
    }
}


// Envía datos a Supabase (SÓLO para Movimiento Brusco)
void enviarDatosSupabase(DateTime now, float at, float T, float safecoin) {
    if (WiFi.status() == WL_CONNECTED) {
        HTTPClient http;
        http.begin(supabaseUrl); // Usa la URL de Supabase
        
        // Headers de Supabase para la inserción
        http.addHeader("Content-Type", "application/json");
        http.addHeader("apikey", supabaseAnonKey); 
        http.addHeader("Authorization", "Bearer " + String(supabaseAnonKey)); // Auth necesaria
        
        // Formatear fecha y hora
        String fecha = String(now.year()) + "-" + 
                       (now.month() < 10 ? "0" : "") + String(now.month()) + "-" +
                       (now.day() < 10 ? "0" : "") + String(now.day());
        
        String hora = (now.hour() < 10 ? "0" : "") + String(now.hour()) + ":" +
                      (now.minute() < 10 ? "0" : "") + String(now.minute()) + ":" +
                      (now.second() < 10 ? "0" : "") + String(now.second());
        
        // Crear JSON con ArduinoJson
        const size_t CAPACITY = JSON_OBJECT_SIZE(8); 
        StaticJsonDocument<CAPACITY> doc;

        doc["at"] = serialized(String(at, 2));        
        doc["T"] = serialized(String(T, 1));          
        doc["fecha"] = fecha;                         
        doc["hora"] = hora;                           
        doc["timestamp"] = now.unixtime();
        doc["latitud"] = serialized(String(latitud, 7));
        doc["longitud"] = serialized(String(longitud, 7));
        doc["safecoin"] = serialized(String(safecoin, 2));

        // Supabase REST API espera un array de objetos JSON
        String singleJson;
        serializeJson(doc, singleJson);
        String finalJsonPayload = "[" + singleJson + "]"; // Envuelve el objeto en un array

        
        Serial.println("\nEnviando a Supabase (Movimiento Brusco): " + finalJsonPayload);
        
        int code = http.POST(finalJsonPayload);
        if (code >= 200 && code < 300) { // 201 Created es el esperado
            Serial.println("Datos enviados a SUPABASE OK - Codigo: " + String(code));
        } else {
            Serial.println("Error al enviar a SUPABASE - Codigo: " + String(code));
            Serial.println("Respuesta del servidor: " + http.getString());
        }
        
        http.end();
    } else {
        Serial.println("WiFi desconectado - No se enviaron datos a Supabase");
    }
}


// Modo Silencioso: Lee, envía a LOCAL y envía a SUPABASE si hay movimiento brusco
void serialOff() {
    DateTime now = rtc.now();
    sensors_event_t a, g, temp;
    mpu.getEvent(&a, &g, &temp);
    
    float ax = a.acceleration.x / 9.81;
    float ay = a.acceleration.y / 9.81;
    float az = a.acceleration.z / 9.81;
    float aceleracionTotal = sqrt(ax*ax + ay*ay + az*az);
    
    // Determinar valor de safecoin según la aceleración
    float safecoin;
    if (aceleracionTotal > 2.0) {
        safecoin = 0.0; // Movimiento brusco
        Serial.println("Movimiento brusco detectado (modo silencioso). Enviando datos a Supabase...");
        // Envío CONDICIONAL a Supabase con safecoin = 0
        enviarDatosSupabase(now, aceleracionTotal, temp.temperature, safecoin); 
    } else {
        safecoin = 0.01; // Condición normal
    }
    
    // Envío CONSTANTE al servidor local con el valor correspondiente de safecoin
    enviarDatosLocal(now, aceleracionTotal, temp.temperature, safecoin); 
    
    delay(500);
}

// Modo Verboso: Imprime, envía a LOCAL y envía a SUPABASE si hay movimiento brusco
void serialOn() {
    // ========== Leer RTC DS3231 ==========
    DateTime now = rtc.now();
    
    Serial.println("\n--- FECHA Y HORA ---");
    Serial.print("Fecha: "); Serial.print(daysOfTheWeek[now.dayOfTheWeek()]); Serial.print(", ");
    if (now.day() < 10) Serial.print('0'); Serial.print(now.day()); Serial.print('/');
    if (now.month() < 10) Serial.print('0'); Serial.print(now.month()); Serial.print('/'); Serial.println(now.year());
    
    Serial.print("Hora: ");
    if (now.hour() < 10) Serial.print('0'); Serial.print(now.hour()); Serial.print(':');
    if (now.minute() < 10) Serial.print('0'); Serial.print(now.minute()); Serial.print(':');
    if (now.second() < 10) Serial.print('0'); Serial.println(now.second());
    
    Serial.print("Temperatura RTC: "); Serial.print(rtc.getTemperature(), 1); Serial.println(" C");
    
    // ========== Leer MPU6050 ==========
    sensors_event_t a, g, temp;
    mpu.getEvent(&a, &g, &temp);
    
    float ax = a.acceleration.x / 9.81; // Convertir m/s² a g
    float ay = a.acceleration.y / 9.81;
    float az = a.acceleration.z / 9.81;
    
    float gx = g.gyro.x * 57.2958; // Convertir rad/s a °/s
    float gy = g.gyro.y * 57.2958;
    float gz = g.gyro.z * 57.2958;
    
    float aceleracionTotal = sqrt(ax*ax + ay*ay + az*az);
    
    Serial.println("\n--- ACELERACION TOTAL (g) ---"); Serial.println(aceleracionTotal, 2);
    
    Serial.println("\n--- GIROSCOPIO (grados/s) ---");
    Serial.print("X: "); Serial.print(gx, 2);
    Serial.print(" | Y: "); Serial.print(gy, 2);
    Serial.print(" | Z: "); Serial.println(gz, 2);
    
    Serial.print("Aceleracion Total: ");
    Serial.print(aceleracionTotal, 2);
    Serial.print(" g");

    // Determinar valor de safecoin según la aceleración
    float safecoin;
    if (aceleracionTotal > 2.0) {
        safecoin = 0.0; // Movimiento brusco
        Serial.println(" - MOVIMIENTO BRUSCO! -> ENVIANDO DATOS A SUPABASE...");
        Serial.print("SafeCoin: ");
        Serial.println(safecoin, 2);
        // Envío CONDICIONAL a Supabase con safecoin = 0
        enviarDatosSupabase(now, aceleracionTotal, temp.temperature, safecoin); 
    } else {
        safecoin = 0.01; // Condición normal
        Serial.println(" - Estable");
        Serial.print("SafeCoin: ");
        Serial.println(safecoin, 2);
    }
    
    // Envío CONSTANTE al servidor local con el valor correspondiente de safecoin
    enviarDatosLocal(now, aceleracionTotal, temp.temperature, safecoin); 
    
    Serial.print("Temperatura MPU: "); Serial.print(temp.temperature, 1); Serial.println(" C");
    
    Serial.println("=================================");
    
    delay(500);
}
